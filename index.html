  <script>

    // 상태 관리

    let idolName = '';


    // DOM 요소

    const idolNameInput = document.getElementById('idol-name');

    const nameLenDisplay = document.getElementById('name-len');

    const displayTitle = document.getElementById('chart-display-title');

    const captureTarget = document.getElementById('chart-capture-target');

    const slotBoxes = document.querySelectorAll('.slot-box');

    const btnDownload = document.getElementById('btn-download');

    const btnReset = document.getElementById('btn-reset');

    const toast = document.getElementById('status-toast');

    const toastMsg = document.getElementById('toast-message');


    // 토스트 알림 함수

    function showToast(message, isError = false) {

      toastMsg.textContent = message;

      const icon = toast.querySelector('i');

      if (isError) {

        icon.className = 'fa-solid fa-circle-exclamation text-rose-400';

      } else {

        icon.className = 'fa-solid fa-circle-check text-purple-400';

      }

      toast.classList.remove('opacity-0', 'pointer-events-none', 'translate-y-2');

      toast.classList.add('opacity-100', 'translate-y-0');


      setTimeout(() => {

        toast.classList.add('opacity-0', 'pointer-events-none', 'translate-y-2');

        toast.classList.remove('opacity-100', 'translate-y-0');

      }, 2200);

    }


    // 이름 입력 시 실시간 타이틀 갱신 ({이름} 취향표 로 고정)

    function updateTitle() {

      const name = idolNameInput.value.trim();

      idolName = name;

      displayTitle.textContent = `${name || '최애'} 취향표`;

      nameLenDisplay.textContent = `${idolNameInput.value.length}/12`;

    }


    idolNameInput.addEventListener('input', updateTitle);


    // 각 슬롯 이미지 업로드 및 동적 렌더링

    slotBoxes.forEach(slot => {

      const fileInput = slot.querySelector('.file-input');

      const placeholder = slot.querySelector('.slot-placeholder');

      const btnRemove = slot.querySelector('.btn-remove-slot');


      // 박스 탭 시 파일 선택기 실행

      slot.addEventListener('click', (e) => {

        if (e.target.closest('.btn-remove-slot')) return;

        fileInput.click();

      });


      // 파일 선택 시 이미지 생성 및 장착

      fileInput.addEventListener('change', (e) => {

        const file = e.target.files[0];

        if (!file) return;


        if (!file.type.startsWith('image/')) {

          showToast('이미지 파일만 첨부할 수 있습니다.', true);

          return;

        }


        const reader = new FileReader();

        reader.onload = (event) => {

          // 기존 이미지가 있다면 제거

          let existingImg = slot.querySelector('img');

          if (!existingImg) {

            existingImg = document.createElement('img');

            slot.insertBefore(existingImg, btnRemove);

          }

          existingImg.src = event.target.result;

          existingImg.alt = "업로드 이미지";

          placeholder.classList.add('hidden');

          btnRemove.classList.remove('hidden');

        };

        reader.readAsDataURL(file);

      });


      // 개별 삭제 버튼

      btnRemove.addEventListener('click', (e) => {

        e.stopPropagation();

        fileInput.value = '';

        const img = slot.querySelector('img');

        if (img) img.remove();

        placeholder.classList.remove('hidden');

        btnRemove.classList.add('hidden');

      });

    });


    // 전체 비우기 버튼

    btnReset.addEventListener('click', () => {

      slotBoxes.forEach(slot => {

        const fileInput = slot.querySelector('.file-input');

        const img = slot.querySelector('img');

        if (img) img.remove();

        fileInput.value = '';

        const placeholder = slot.querySelector('.slot-placeholder');

        const btnRemove = slot.querySelector('.btn-remove-slot');

        placeholder.classList.remove('hidden');

        btnRemove.classList.add('hidden');

      });

      showToast('사진이 모두 비워졌습니다.');

    });


    // 고해상도 PNG 저장 처리

    async function downloadChartImage() {

      const originalText = btnDownload.innerHTML;

      btnDownload.disabled = true;

      btnDownload.innerHTML = `<i class="fa-solid fa-spinner fa-spin text-lg"></i> <span>이미지 생성 중...</span>`;


      // 캡처 전 우측 상단 X 버튼 일시 숨김

      const removeButtons = document.querySelectorAll('.btn-remove-slot');

      removeButtons.forEach(btn => btn.style.display = 'none');


      try {

        const dataUrl = await htmlToImage.toPng(captureTarget, {

          quality: 1.0,

          pixelRatio: 2.5,

          cacheBust: true,

          style: {

            transform: 'none',

            margin: '0'

          }

        });


        const safeName = (idolName || '최애').replace(/\s+/g, '_');

        const link = document.createElement('a');

        link.download = `${safeName}_취향표.png`;

        link.href = dataUrl;

        document.body.appendChild(link);

        link.click();

        document.body.removeChild(link);


        showToast('취향표가 성공적으로 저장되었습니다! 💜');

      } catch (err) {

        console.error('저장 실패:', err);

        showToast('저장 중 오류가 발생했습니다. 다시 시도해주세요.', true);

      } finally {

        removeButtons.forEach(btn => btn.style.display = '');

        btnDownload.disabled = false;

        btnDownload.innerHTML = originalText;

      }

    }


    btnDownload.addEventListener('click', downloadChartImage);

  </script>
